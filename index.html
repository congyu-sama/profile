1. UDP Client — udpclient.java
import java.io.*;
import java.net.*;

public class udpclient {
    public static void main(String [] args){
        try{
            DatagramSocket ds=new DatagramSocket();
            InetAddress server=InetAddress.getByName("127.0.0.1");
            while(true){
                InputStreamReader isr=new InputStreamReader(System.in);
                BufferedReader br=new BufferedReader(isr);
                System.out.print("Nhap chuoi: ");
                String s=br.readLine();
                if (s.equalsIgnoreCase("exit")) break;
                byte[] data=s.getBytes();
                DatagramPacket dp=new DatagramPacket(data,data.length,server,7);
                ds.send(dp);
                byte[] buffer=new byte[256];
                DatagramPacket incoming=new DatagramPacket(buffer,buffer.length);
                ds.receive(incoming);
                System.out.println("Server: "+new String(incoming.getData(),0,incoming.getLength()));
            }
        }catch(IOException e){
            System.out.println(""+e);
        }
    }
}
2. UDP Server — udpserver.java
import java.io.*;
import java.net.*;

public class udpserver {
    public static void main (String[] args){
        try{
            DatagramSocket ds=new DatagramSocket(7);
            System.out.println("Server cho 7...");
            byte[] buffer=new byte[256];
            while(true){
                DatagramPacket in=new DatagramPacket(buffer,buffer.length);
                ds.receive(in);
                String str=new String(in.getData(),0,in.getLength());
                System.out.println("Client: "+str);
                str=str.toUpperCase();
                byte[] data=str.getBytes();
                DatagramPacket out=new DatagramPacket(data,in.getLength(),in.getAddress(),in.getPort());
                ds.send(out);
            }
        }catch(IOException e){
            System.out.println(""+e);
        }
    }
}
3. TCP Client — tcpclient.java
import java.io.*;
import java.net.*;
import java.util.Scanner;

public class tcpclient {
    public static void main(String[] args){
        try{
            Socket socket=new Socket("localhost",5000);
            DataInputStream in=new DataInputStream(socket.getInputStream());
            DataOutputStream out=new DataOutputStream(socket.getOutputStream());
            System.out.println("Nhap n=");
            Scanner sc=new Scanner(System.in);
            int n=sc.nextInt();
            out.writeInt(n); out.flush();
            int tong=in.readInt();
            System.out.println("Tong "+tong);
        }catch(IOException e){
            System.out.println(""+e);
        }
    }
}
4. TCP Server — tcpserver.java
import java.io.*;
import java.net.*;

public class tcpserver {
    public static void main(String [] args)throws Exception{
        ServerSocket ss=new ServerSocket(5000);
        System.out.println("Server cho 5000...");
        while(true){
            Socket cs=ss.accept();
            DataInputStream in=new DataInputStream(cs.getInputStream());
            DataOutputStream out=new DataOutputStream(cs.getOutputStream());
            int n=in.readInt();
            System.out.println("Client: "+n);
            int tong=0;
            for (int i=1; i<=n; i++){
                tong+=i;
            }
            out.writeInt(tong); out.flush();
        }
    }
}
5. UDP Multicast Client — udpMulticastClient.java
import java.io.*;
import java.net.*;

public class udpMulticastClient {
    public static void main(String[] args) {
        try {
            DatagramSocket ds = new DatagramSocket();
            InetAddress multicastAddress = InetAddress.getByName("230.0.0.1");
            BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
            while (true) {
                System.out.print("Nhap chuoi: ");
                String s = br.readLine();
                if (s.equalsIgnoreCase("exit")) {break;}
                byte[] data = s.getBytes();
                DatagramPacket dp = new DatagramPacket(data,data.length,multicastAddress,9013);
                ds.send(dp);
                byte[] buffer = new byte[256];
                DatagramPacket incoming = new DatagramPacket(buffer,buffer.length);
                ds.receive(incoming);
                String result = new String(incoming.getData(),0,incoming.getLength());
                System.out.println("Server: " + result);
            }
            ds.close();
        } catch (IOException e) {
            System.out.println("Loi: " + e);
        }
    }
}
6. UDP Multicast Server — udpMulticastServer.java
import java.io.*;
import java.net.*;

public class udpMulticastServer {
    public static void main(String[] args) {
        try {
            MulticastSocket ds = new MulticastSocket(9013);
            InetAddress address = InetAddress.getByName("230.0.0.1");
            ds.joinGroup(address);
            System.out.println("Server cho 7...");
            while (true) {
                byte[] buffer = new byte[256];
                DatagramPacket in = new DatagramPacket(buffer,buffer.length);
                ds.receive(in);
                String str = new String(in.getData(),0,in.getLength());
                System.out.println("Client: " + str);
                str = str.toUpperCase();
                byte[] data = str.getBytes();
                DatagramPacket out = new DatagramPacket(data,data.length,in.getAddress(),in.getPort());
                ds.send(out);
            }
        } catch (IOException e) {
            System.out.println("Loi: " + e);
        }
    }
}
